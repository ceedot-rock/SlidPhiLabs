function encodeWithANS(n) {
  const pos  = zeckendorf(n);
  const gaps = [pos[0], ...pos.slice(1).map((p,i) => p - pos[i])];
  // hybrid path (slid-phi): one 11 + terminator 111 + low-bits binary
  // gap+ANS path: rANS on gaps ≈1.2 bits/gap when P(g=2)≈0.5
  return { pos, gaps };
}
